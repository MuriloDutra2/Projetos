import { useDialog } from '../providers/DialogProvider'
import { friendlyError } from '../utils/errorHandler'

interface DailyReportData {
  totalAvulsos: number
  planosVendidosCount: number
  planosVendidosValue: number
  saved: { qtyCars: number; qtyMotos: number; createdAt: string } | null
}

export default function Relatorio(): React.JSX.Element {
  const { showAlert } = useDialog()
  const [reportDay, setReportDay] = useState<string>(() => format(new Date(), 'yyyy-MM-dd'))
  const [dailyReport, setDailyReport] = useState<DailyReportData | null>(null)

  useEffect(() => {
    getDailyReport(reportDay).then(setDailyReport).catch(console.error)
  }, [reportDay])

  const handleSave = async (): Promise<void> => {
    if (!dailyReport) return
    try {
      // Re-fetch tickets at click time (Open Question 1 resolution): simpler than prop drilling
      const currentTickets = await getTickets()
      const qtyCars = currentTickets.filter((t) => t.tipo === 'Carro' || t.tipo === 'MENSAL_CARRO' || t.tipo === 'GARAGEM').length
      const qtyMotos = currentTickets.filter((t) => t.tipo === 'Moto' || t.tipo === 'MENSAL_MOTO').length
      // Verificar com App.tsx:1077 a regra exata de tipo — copiar verbatim a expressão usada no original
      const res = await saveDailyReport({
        dateStr: reportDay,
        totalAvulsos: dailyReport.totalAvulsos,
        planosVendidosCount: dailyReport.planosVendidosCount,
        planosVendidosValue: dailyReport.planosVendidosValue,
        qtyCars,
        qtyMotos
      })
      if (res.success) {
        showAlert('Sucesso', 'Relatório salvo com sucesso!', 'success')
        // Re-fetch o relatório para atualizar a flag `saved`
        const updated = await getDailyReport(reportDay)
        setDailyReport(updated)

        showAlert('Erro', friendlyError(res.error ?? 'Não foi possível salvar.'), 'error')

  const [history, setHistory] = useState<HistoryEntry[]>([])

          value: p.amount ?? 0
        }))
    ].sort((a, b) => new Date(b.date).getTime() - new Date(a.date).getTime())
  }, [history, financialHistory, financeFilterMonth, financeFilterYear])

      }
    } catch (err) {

    }
