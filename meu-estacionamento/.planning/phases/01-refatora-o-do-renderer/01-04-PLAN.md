  toggleClientStatus,
  deleteClient,
  getClientStatement
} from '../services/clients'

import ModalRenovar from '../components/ModalRenovar'
import { useDialog } from '../providers/DialogProvider'
import { friendlyError } from '../utils/errorHandler'
import { formatCpf, formatPhone, maskPlate } from '../utils/masks'
import type { ClientRow, ClientStatement } from '../types/domain'

// Imperative handle interface — App pode pedir abrir o modal de novo cliente via ref (Ctrl+N)
export interface MensalistasHandle {
  openNewClientModal: () => void
}

const Mensalistas = forwardRef<MensalistasHandle>((_props, ref): React.JSX.Element => {
  const { showAlert, showConfirm } = useDialog()

  const [clients, setClients] = useState<ClientRow[]>([])
  const [searchMensalistas, setSearchMensalistas] = useState<string>('')
  const [modalNovoClienteOpen, setModalNovoClienteOpen] = useState<boolean>(false)
  const [clientToEdit, setClientToEdit] = useState<ClientToEdit | null>(null)
  const [modalRenovarOpen, setModalRenovarOpen] = useState<boolean>(false)
  const [renovarClient, setRenovarClient] = useState<ClientRow | null>(null)
  const [statementOpen, setStatementOpen] = useState<boolean>(false)
  const [statementData, setStatementData] = useState<ClientStatement | null>(null)
  const [deleteClientModal, setDeleteClientModal] = useState<{ open: boolean; client: ClientRow | null }>({ open: false, client: null })
  const [deleteClientPassword, setDeleteClientPassword] = useState<string>('')
  const [deleteClientError, setDeleteClientError] = useState<string>('')
  const [deleteClientLoading, setDeleteClientLoading] = useState<boolean>(false)

  const loadClients = useCallback(async (): Promise<void> => {
    try {
      const data = await getClients()
      setClients(data)
    } catch (e) {
      console.error(e)
      setClients([])
      showAlert('Erro', 'Erro ao carregar mensalistas. Tente novamente.', 'error')
    }
  }, [showAlert])

  useEffect(() => {
    loadClients()
  }, [loadClients])

  // Imperative handle: App.tsx chama ref.current?.openNewClientModal() no Ctrl+N
  useImperativeHandle(ref, () => ({
    openNewClientModal: () => {
      setClientToEdit(null)
      setModalNovoClienteOpen(true)
    }
  }), [])

  const openEditModal = (c: ClientRow): void => {
    setClientToEdit({
      id: c.id,
      name: c.name,
      cpf: c.cpf ?? '',
      phone: c.phone ?? '',
      plan_type: c.plan_type,
      expiry_date: c.expiry_date,
      plates: c.plates,
      garage_billing_day: c.garage_billing_day ?? null,
      garage_billing_month: c.garage_billing_month ?? null
    })
    setModalNovoClienteOpen(true)
  }

  const openCancelConfirm = (c: ClientRow): void => {
    showConfirm(
      'Cancelar plano',
      `Deseja cancelar o plano de ${c.name}? O cliente perderá o acesso imediato.`,
      async () => {
        const res = await toggleClientStatus({ clientId: c.id, active: 0 })
        if (res.success) loadClients()
        else showAlert('Erro', friendlyError(res.error ?? 'Não foi possível cancelar'), 'error')
      }
    )
  }

  const openReativarConfirm = (c: ClientRow): void => {
    showConfirm(
      'Reativar plano',
      `Deseja reativar o plano de ${c.name}?`,
      async () => {
        const res = await toggleClientStatus({ clientId: c.id, active: 1 })
        if (res.success) loadClients()
        else showAlert('Erro', friendlyError(res.error ?? 'Não foi possível reativar'), 'error')
      }
    )
  }

  const openStatement = async (c: ClientRow): Promise<void> => {
    try {
      const data = await getClientStatement(c.id)
      setStatementData(data)
      setStatementOpen(true)
    } catch (err) {
      console.error(err)
      showAlert('Erro', friendlyError(err), 'error')
    }
  }

  const openRenovarFor = (c: ClientRow): void => {
    setRenovarClient(c)
    setModalRenovarOpen(true)
  }

  const openDeleteModal = (c: ClientRow): void => {
    setDeleteClientModal({ open: true, client: c })
    setDeleteClientPassword('')
    setDeleteClientError('')
  }

  const handleDeleteConfirm = async (): Promise<void> => {
    if (!deleteClientModal.client) return
    setDeleteClientLoading(true)
    setDeleteClientError('')
    try {
      const res = await deleteClient({ clientId: deleteClientModal.client.id, password: deleteClientPassword })
      if (res.success) {
        setDeleteClientModal({ open: false, client: null })
        setDeleteClientPassword('')
        loadClients()
        showAlert('Sucesso', 'Mensalista excluído com sucesso.', 'success')
      } else {
        setDeleteClientError(res.error ?? 'Senha incorreta')
      }
    } catch (err) {
      setDeleteClientError(friendlyError(err))
    } finally {
      setDeleteClientLoading(false)
    }
  }

  // Filtragem client-side por busca (replicar lógica original — verificar App.tsx exato)
  const filteredClients = clients.filter((c) => {
    if (!searchMensalistas) return true
    const q = searchMensalistas.toLowerCase()
    return (
      c.name.toLowerCase().includes(q) ||
      c.plates.some((p) => p.toLowerCase().includes(q)) ||
      (c.cpf?.toLowerCase().includes(q) ?? false)
    )
  })

  return (
    <div className="flex-1 p-6 overflow-y-auto">
      {/* JSX completo de App.tsx:1135-1336 — copiar verbatim:
          - Header com botão "Novo Cliente" (onClick: abrir modal com clientToEdit=null) e busca
          - Cards/lista de clientes (mapear filteredClients) com:
              - Nome, CPF formatado, plan_type, expiry_date, status (Em dia/Em atraso/etc.), plates
              - Botões: Editar, Cancelar/Reativar, Renovar, Extrato, Excluir
          - Empty state quando filteredClients.length === 0
          - Strings em pt-BR exatas
      */}

      <ModalNovoCliente
        open={modalNovoClienteOpen}
        onClose={() => {
          setModalNovoClienteOpen(false)
          setClientToEdit(null)
        }}
        onSuccess={loadClients}
        onAlert={showAlert}
        clientToEdit={clientToEdit}
      />

      <ModalRenovar
        open={modalRenovarOpen}
        onClose={() => {
          setModalRenovarOpen(false)
          setRenovarClient(null)
        }}
        onSuccess={loadClients}
        onAlert={showAlert}
        client={renovarClient}
      />

      {/* deleteClientModal inline — copiar verbatim de App.tsx:1637-1705:
          - if (!deleteClientModal.open) return null OU render condicional via &&
          - input type="password" value={deleteClientPassword}
          - botão Confirmar onClick={handleDeleteConfirm}
          - botão Cancelar onClick={() => setDeleteClientModal({ open: false, client: null })}
          - mostrar deleteClientError quando truthy
          - disable botões com deleteClientLoading
      */}

      {/* statement modal inline — copiar verbatim de App.tsx:1794-1834:
          - if (!statementOpen || !statementData) return null
          - render dos statementData.payments e statementData.avulsoWhileDebtor
          - botão Fechar
      */}
    </div>
  )
})

Mensalistas.displayName = 'Mensalistas'

export default Mensalistas
